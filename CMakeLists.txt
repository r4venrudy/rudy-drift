cmake_minimum_required(VERSION 3.20)
project(RudyDrift VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CTest)
include(CMakePackageConfigHelpers)
option(RD_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)

add_library(rudy_drift STATIC src/forest.c)
add_library(RudyDrift::rudy_drift ALIAS rudy_drift)
target_include_directories(rudy_drift PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_features(rudy_drift PUBLIC c_std_17)
set_target_properties(rudy_drift PROPERTIES C_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
if(NOT MSVC)
  target_link_libraries(rudy_drift PUBLIC m)
endif()
add_executable(rudy-drift src/main.c src/data.c)
target_link_libraries(rudy-drift PRIVATE rudy_drift)

function(rd_checks target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
  endif()
  if(RD_SANITIZERS AND NOT MSVC)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endfunction()
rd_checks(rudy_drift)
rd_checks(rudy-drift)

if(BUILD_TESTING)
  add_executable(test-forest tests/test_forest.c)
  target_link_libraries(test-forest PRIVATE rudy_drift)
  rd_checks(test-forest)
  add_test(NAME forest COMMAND test-forest)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:rudy-drift>)
  add_test(NAME benchmark-metrics COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_metrics.py)
  set_tests_properties(cli PROPERTIES TIMEOUT 120)
endif()

install(TARGETS rudy_drift rudy-drift EXPORT RudyDriftTargets
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/rudy_drift.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT RudyDriftTargets NAMESPACE RudyDrift:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/RudyDrift)
configure_package_config_file(cmake/RudyDriftConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/RudyDriftConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/RudyDrift)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/RudyDriftConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/RudyDriftConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/RudyDriftConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/RudyDrift)
install(FILES README.md LICENSE SECURITY.md CONTRIBUTING.md DESTINATION ${CMAKE_INSTALL_DATADIR}/rudy-drift)
install(DIRECTORY docs examples assets benchmarks DESTINATION ${CMAKE_INSTALL_DATADIR}/rudy-drift PATTERN "__pycache__" EXCLUDE)
set(CPACK_PACKAGE_NAME rudy-drift)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_GENERATOR ZIP)
include(CPack)
